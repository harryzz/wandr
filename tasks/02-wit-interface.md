```wit
package my:skiko-gfx@0.1.0;

/// Drawing commands imported by the Kotlin WASM guest.
/// Implemented on the Rust host side by skia-safe.
interface canvas {
    // --- Surface ---
    surface-width:  func() -> u32;
    surface-height: func() -> u32;
    begin-frame:    func();
    end-frame:      func();   // flush + swap buffers

    // --- Transform stack ---
    save:      func();
    restore:   func();
    translate: func(dx: f32, dy: f32);
    scale:     func(sx: f32, sy: f32);
    rotate:    func(degrees: f32);

    // --- Clip ---
    clip-rect: func(x: f32, y: f32, w: f32, h: f32, anti-alias: bool);

    // --- Drawing ---
    clear:      func(argb: u32);
    draw-rect:  func(x: f32, y: f32, w: f32, h: f32, paint: paint-attrs);
    draw-rrect: func(x: f32, y: f32, w: f32, h: f32,
                     rx: f32, ry: f32, paint: paint-attrs);
    draw-oval:  func(x: f32, y: f32, w: f32, h: f32, paint: paint-attrs);
    draw-line:  func(x0: f32, y0: f32, x1: f32, y1: f32, paint: paint-attrs);
    draw-path:  func(cmds: list<u8>, paint: paint-attrs);

    // --- Resources ---
    /// Upload RGBA8888 pixels; returns opaque host image ID.
    create-image:  func(width: u32, height: u32, pixels: list<u8>) -> u32;
    draw-image:    func(id: u32, x: f32, y: f32, alpha: u8);
    drop-image:    func(id: u32);

    /// Create a text blob; returns opaque host ID.
    create-text-blob: func(text: string, font-family: string,
                           size: f32, weight: u32) -> u32;
    draw-text-blob:   func(id: u32, x: f32, y: f32, paint: paint-attrs);
    drop-text-blob:   func(id: u32);
}

record paint-attrs {
    color:        u32,   // ARGB packed
    style:        paint-style,
    stroke-width: f32,
    anti-alias:   bool,
    alpha:        u8,
}

enum paint-style { fill, stroke, fill-and-stroke }

/// Functions exported by the Kotlin WASM guest.
/// The Rust host calls these each vsync / on input.
interface renderer {
    render-frame:     func(nanos: u64);
    on-pointer-event: func(kind: pointer-kind, x: f32, y: f32);
    on-key-event:     func(kind: key-kind, key-code: u32);
    on-resize:        func(w: u32, h: u32);
}

enum pointer-kind { down, up, move, scroll }
enum key-kind     { down, up }

world skiko-ui {
    import canvas;
    export renderer;
}
```

```rust
// host/src/canvas_impl.rs

use anyhow::Result;
use skia_safe::{Canvas, Color, Paint, PaintStyle, Path, Rect, RRect, Surface};
use std::sync::Arc;
use winit::window::Window;

// Generated by bindgen! — import the WIT types
use crate::bindings::my::skiko_gfx::canvas::{PaintAttrs, PaintStyle as WitPaintStyle};

// Resource tables
pub struct ResourceTable {
    images: slab::Slab<skia_safe::Image>,
    blobs:  slab::Slab<skia_safe::TextBlob>,
}

impl ResourceTable {
    pub fn new() -> Self {
        Self {
            images: slab::Slab::new(),
            blobs:  slab::Slab::new(),
        }
    }
}

pub struct SkiaRenderer {
    #[cfg(target_os = "android")]
    egl:        crate::egl::android::EglContext,
    #[cfg(target_os = "android")]
    gr_context: skia_safe::gpu::DirectContext,

    surface:    Surface,
    pub width:  u32,
    pub height: u32,
    pub resources: ResourceTable,
}

impl SkiaRenderer {
    pub fn new(window: Arc<Window>) -> Result<Self> {
        let size = window.inner_size();

        #[cfg(target_os = "android")]
        {
            use winit::platform::android::WindowExtAndroid;
            let native_window = window.a_native_window() as *mut std::ffi::c_void;
            let egl = crate::egl::android::EglContext::new(native_window)?;

            let gl_interface = skia_safe::gpu::gl::Interface::new_load_with(
                crate::egl::android::EglContext::proc_resolver()
            ).ok_or_else(|| anyhow::anyhow!("GL interface failed"))?;

            let mut gr_context = skia_safe::gpu::direct_contexts::make_gl(
                gl_interface, None
            ).ok_or_else(|| anyhow::anyhow!("GrContext failed"))?;

            let surface = Self::make_gl_surface(
                &mut gr_context, egl.width, egl.height)?;

            return Ok(Self {
                egl, gr_context, surface,
                width: size.width, height: size.height,
                resources: ResourceTable::new(),
            });
        }

        #[cfg(not(target_os = "android"))]
        {
            let surface = skia_safe::surfaces::raster_n32_premul(
                (size.width as i32, size.height as i32)
            ).ok_or_else(|| anyhow::anyhow!("raster surface failed"))?;
            Ok(Self {
                surface, width: size.width, height: size.height,
                resources: ResourceTable::new(),
            })
        }
    }

    #[cfg(target_os = "android")]
    fn make_gl_surface(
        gr: &mut skia_safe::gpu::DirectContext,
        w: i32, h: i32,
    ) -> Result<Surface> {
        let fb_info = skia_safe::gpu::gl::FramebufferInfo {
            fboid:  0,
            format: skia_safe::gpu::gl::Format::RGBA8.into(),
            ..Default::default()
        };
        let target = skia_safe::gpu::backend_render_targets::make_gl(
            (w, h), None, 8, fb_info);
        skia_safe::gpu::surfaces::wrap_backend_render_target(
            gr, &target,
            skia_safe::gpu::SurfaceOrigin::BottomLeft,
            skia_safe::ColorType::RGBA8888,
            None, None,
        ).ok_or_else(|| anyhow::anyhow!("wrap_backend_render_target failed"))
    }

    pub fn canvas(&mut self) -> &Canvas {
        self.surface.canvas()
    }

    pub fn flush_and_swap(&mut self) {
        self.surface.flush_and_submit();
        #[cfg(target_os = "android")]
        self.egl.swap();
    }

    pub fn resize(&mut self, w: u32, h: u32) {
        self.width  = w;
        self.height = h;
        #[cfg(target_os = "android")]
        {
            // Recreate surface at new size
            if let Ok(s) = Self::make_gl_surface(
                &mut self.gr_context, w as i32, h as i32)
            {
                self.surface = s;
            }
        }
        #[cfg(not(target_os = "android"))]
        {
            if let Some(s) = skia_safe::surfaces::raster_n32_premul(
                (w as i32, h as i32)) {
                self.surface = s;
            }
        }
    }

    /// Test frame — used in Task 01 stub, kept for debug
    pub fn draw_test_frame(&mut self) {
        let c = self.surface.canvas();
        c.clear(Color::from_argb(255, 10, 20, 60));
        c.draw_rect(
            Rect::from_xywh(50.0, 50.0, 200.0, 100.0),
            &Paint::new(skia_safe::Color4f::new(1.0, 1.0, 1.0, 1.0), None),
        );
        self.flush_and_swap();
    }
}

// ─── WIT canvas trait implementation ─────────────────────────────────────────

impl crate::bindings::my::skiko_gfx::canvas::Host for crate::HostState {

    fn surface_width(&mut self)  -> u32 { self.renderer.width }
    fn surface_height(&mut self) -> u32 { self.renderer.height }

    fn begin_frame(&mut self) { /* surface already acquired */ }

    fn end_frame(&mut self) {
        self.renderer.flush_and_swap();
    }

    fn save(&mut self)    { self.renderer.canvas().save(); }
    fn restore(&mut self) { self.renderer.canvas().restore(); }

    fn translate(&mut self, dx: f32, dy: f32) {
        self.renderer.canvas().translate((dx, dy));
    }
    fn scale(&mut self, sx: f32, sy: f32) {
        self.renderer.canvas().scale((sx, sy));
    }
    fn rotate(&mut self, degrees: f32) {
        self.renderer.canvas().rotate(degrees, None);
    }

    fn clip_rect(&mut self, x: f32, y: f32, w: f32, h: f32, anti_alias: bool) {
        self.renderer.canvas().clip_rect(
            Rect::from_xywh(x, y, w, h),
            None, anti_alias,
        );
    }

    fn clear(&mut self, argb: u32) {
        self.renderer.canvas().clear(Color::from(argb));
    }

    fn draw_rect(&mut self, x: f32, y: f32, w: f32, h: f32, p: PaintAttrs) {
        let paint = p.to_skia();
        self.renderer.canvas().draw_rect(Rect::from_xywh(x, y, w, h), &paint);
    }

    fn draw_rrect(&mut self, x: f32, y: f32, w: f32, h: f32,
                   rx: f32, ry: f32, p: PaintAttrs) {
        let paint  = p.to_skia();
        let rrect  = RRect::new_rect_xy(Rect::from_xywh(x, y, w, h), rx, ry);
        self.renderer.canvas().draw_rrect(rrect, &paint);
    }

    fn draw_oval(&mut self, x: f32, y: f32, w: f32, h: f32, p: PaintAttrs) {
        let paint = p.to_skia();
        self.renderer.canvas().draw_oval(Rect::from_xywh(x, y, w, h), &paint);
    }

    fn draw_line(&mut self, x0: f32, y0: f32, x1: f32, y1: f32, p: PaintAttrs) {
        let paint = p.to_skia();
        self.renderer.canvas().draw_line((x0, y0), (x1, y1), &paint);
    }

    fn draw_path(&mut self, cmds: Vec<u8>, p: PaintAttrs) {
        let path  = decode_path(&cmds);
        let paint = p.to_skia();
        self.renderer.canvas().draw_path(&path, &paint);
    }

    fn create_image(&mut self, width: u32, height: u32, pixels: Vec<u8>) -> u32 {
        let info = skia_safe::ImageInfo::new_n32_premul(
            (width as i32, height as i32), None);
        let image = skia_safe::images::raster_from_data(
            &info,
            skia_safe::Data::new_copy(&pixels),
            (width * 4) as usize,
        ).unwrap();
        self.renderer.resources.images.insert(image) as u32
    }

    fn draw_image(&mut self, id: u32, x: f32, y: f32, alpha: u8) {
        if let Some(img) = self.renderer.resources.images.get(id as usize) {
            let mut p = Paint::default();
            p.set_alpha(alpha);
            self.renderer.canvas()
                .draw_image(img, (x, y), Some(&p));
        }
    }

    fn drop_image(&mut self, id: u32) {
        self.renderer.resources.images.try_remove(id as usize);
    }

    fn create_text_blob(&mut self, text: String, font_family: String,
                         size: f32, weight: u32) -> u32 {
        use skia_safe::{Font, Typeface, FontStyle};
        let style    = FontStyle::new(
            weight as i32, 5,
            skia_safe::font_style::Slant::Upright);
        let typeface = Typeface::new(&font_family, style)
            .unwrap_or_default();
        let font     = Font::new(typeface, size);
        let blob     = skia_safe::TextBlob::new(&text, &font).unwrap();
        self.renderer.resources.blobs.insert(blob) as u32
    }

    fn draw_text_blob(&mut self, id: u32, x: f32, y: f32, p: PaintAttrs) {
        if let Some(blob) = self.renderer.resources.blobs.get(id as usize) {
            let paint = p.to_skia();
            self.renderer.canvas().draw_text_blob(blob, (x, y), &paint);
        }
    }

    fn drop_text_blob(&mut self, id: u32) {
        self.renderer.resources.blobs.try_remove(id as usize);
    }
}

// ─── Helpers ─────────────────────────────────────────────────────────────────

impl PaintAttrs {
    pub fn to_skia(&self) -> Paint {
        let mut p = Paint::default();
        p.set_color(Color::from(self.color));
        p.set_style(match self.style {
            WitPaintStyle::Fill         => PaintStyle::Fill,
            WitPaintStyle::Stroke       => PaintStyle::Stroke,
            WitPaintStyle::FillAndStroke => PaintStyle::StrokeAndFill,
        });
        p.set_stroke_width(self.stroke_width);
        p.set_anti_alias(self.anti_alias);
        p.set_alpha(self.alpha);
        p
    }
}

/// Decode the path binary format emitted by Path.wasmWasi.kt
pub fn decode_path(data: &[u8]) -> Path {
    let mut path = Path::new();
    let mut i = 0usize;
    let f = |buf: &[u8], off: usize| -> f32 {
        let b = &buf[off..off + 4];
        f32::from_bits(u32::from_be_bytes([b[0], b[1], b[2], b[3]]))
    };
    while i < data.len() {
        let tag = data[i]; i += 1;
        match tag {
            0x01 => { path.move_to((f(data,i), f(data,i+4))); i+=8; }
            0x02 => { path.line_to((f(data,i), f(data,i+4))); i+=8; }
            0x03 => { path.cubic_to(
                (f(data,i),    f(data,i+4)),
                (f(data,i+8),  f(data,i+12)),
                (f(data,i+16), f(data,i+20))); i+=24; }
            0x04 => { path.close(); }
            _    => break,
        }
    }
    path
}
```

## Known issues

### `my::skiko_gfx::canvas::Host` not found

The `bindgen!` path is relative to the crate root (`Cargo.toml` location),
not to `src/`. Make sure the path is `"../wit/skiko-gfx.wit"` not
`"../../wit/skiko-gfx.wit"`.

### `PaintStyle::StrokeAndFill` not found in skia-safe

Older skia-safe uses `PaintStyle::StrokeAndFill`. Newer uses `Fill`,
`Stroke`, `StrokeAndFill`. Check with:
```bash
cargo doc --open -p skia-safe 2>/dev/null
# or
grep -r "StrokeAndFill" ~/.cargo/registry/src/*/skia-safe-*/src/
```

### `slab` crate not found

Add `slab = "0.4"` to `[dependencies]` in `Cargo.toml`.

## Do NOT

- Do not implement the `renderer` interface in Rust — that is implemented
  by the Kotlin WASM guest. The Rust side only calls it (Task 05+07).
- Do not add actual WASM loading yet — that comes in Task 07.
- Do not change the WIT after this point without regenerating bindings.
